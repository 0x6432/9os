(if timeout -s KILL 20 strace -f -o /tmp/alpine-strace.log /bin/sh -c "echo native-trace-ok; /bin/true" && grep -q execve /tmp/alpine-strace.log; then r=0; else r=$?; fi; [ "$r" -eq 0 ] || (d(){ [ -f /tmp/alpine-strace.log ] || return 1; t=$(mktemp) || return 1; trap 'rm -f "$t" >/dev/null 2>&1 || :' EXIT; tail -c1025 /tmp/alpine-strace.log >"$t" || return 1; n=$(wc -c <"$t") || return 1; case $n in ''|*[!0-9]*|0?*) return 1;; esac; [ "${#n}" -le 4 ] || return 1; [ "$n" -le 1025 ] || return 1; s=0; [ "$n" -le 1024 ] || s=1; n=$((n-s)); h=$(LC_ALL=C od -An -v -tx1 -j "$s" "$t") || return 1; printf %s "$h" >"$t" || return 1; h=$(LC_ALL=C tr -d '[:space:]' <"$t") || return 1; case $h in *[!0-9a-f]*) return 1;; esac; [ "${#h}" -eq "$((n*2))" ] || return 1; rm -f "$t" || return 1; printf 'ALPINE_STRACE_TAIL bytes=%s truncated=%s hex=%s\n' "$n" "$s" "$h"; }; d 2>/dev/null || printf 'ALPINE_STRACE_TAIL unknown\n' ); exit "$r")
